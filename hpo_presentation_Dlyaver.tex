\begin{frame}{Motivation \& Problem Setting}
  \textbf{Goal:} Find the best hyperparameters for a learning algorithm that maximise performance on validation data
  \bigskip
  \twocol{
    \textbf{Why is this hard?}
    \begin{itemize}[<+->]
      \item We treat the model as a black box
      \item Each evaluation requires full model training (expensive)
      \item Search space is large and heterogeneous
            (continuous, integer, categorical, conditional)
    \end{itemize}
  }{
    \textbf{This assignment}
    \begin{itemize}[<+->]
      \item 5 HPO strategies implemented from scratch
            (RS, Grid Search, SH, HB, BO)
      \item Evaluate on 2 YAHPO Gym benchmarks
      \item Fair comparison using the same total budget (fidelity)
    \end{itemize}
  }
\end{frame}

\section{Experimental Design}

\begin{frame}{Experimental Design}
  \begin{itemize}[<+->]
    \item Fair comparison: all methods use the same total budget 
          \[
          \text{B}_{\max} = 50 \times r_{\max}
          \]
    \item Robust evaluation: results averaged over 10 random seeds
    \item Benchmarks: two surrogate tasks from YAHPO Gym
      \begin{itemize}[<+->]
        \item NB301 / CIFAR-10 — 34-dim categorical NAS space
          \begin{itemize}
            \item fidelity: training epochs ($r_{\min}=1$, $r_{\max}=98$)
          \end{itemize}
        \item rbv2\_xgboost / instance 16 — 8-dim mixed space
          \begin{itemize}
            \item fidelity: training set fraction ($r_{\min}=0$, $r_{\max}=1$)
          \end{itemize}
      \end{itemize}
    \item Implementation: ConfigSpace used without forbidden methods
          (\texttt{sample\_configuration}, \texttt{get\_array},
           \texttt{generate\_grid}) — all reimplemented manually
  \end{itemize}
\end{frame}

\section{Algorithms}
% ===================================================================

\begin{frame}{Random Search \citeref{Bergstra \& Bengio, 2012}}
  \begin{itemize}[<+->]
    \item \textbf{Strategy:} sample configurations \alert{uniformly at random};
          evaluate each at full budget $r_{\max}$
    \item \textbf{No learning} from past observations — pure baseline
    \item Gives $B / r_{\max} = 50$ independent full-fidelity evaluations
    \item Surprisingly competitive when most HPs have little effect
          \citeref{Bergstra \& Bengio, 2012}
  \end{itemize}

  \vspace{0.5em}
  \begin{block}{Implementation note}
    Manual sampler over each HP type:
    uniform draws for \texttt{UniformFloat}/\texttt{Integer},
    \texttt{rng.choice} for \texttt{Categorical}.
    Conditional spaces handled via \texttt{get\_active\_hyperparameters()}.
  \end{block}
\end{frame}

\begin{frame}{Grid Search}
  \twocol{
    \textbf{Strategy:} discretise each HP dimension; evaluate all
    grid points at $r_{\max}$.

    \medskip
    \textbf{Discretisation:}
    \begin{itemize}
      \item \texttt{UniformFloat/Integer} $\to$ 5 linearly-spaced values
      \item \texttt{Categorical/Ordinal} $\to$ all choices
    \end{itemize}

    \medskip
    \textbf{Cartesian explosion:} NB301 has 34 categorical HPs
    (${\sim}8$ ops each) $\leadsto$ $8^{34}$ combinations.
    \alert{Infeasible to enumerate.}
  }{
    \textbf{Two-path implementation}
    \begin{itemize}
      \item Grid $\leq 10{,}000$ configs $\to$ full enumeration
      \item Grid $> 10{,}000$ configs \emph{or} conditional space
            $\to$ random subsampling of unique combos
    \end{itemize}

    \vspace{0.4em}
    \begin{alertblock}{NB301 outcome}
      24 active conditions collapse the grid to random sampling
      $\leadsto$ GS $\approx$ RS on NB301 by construction
    \end{alertblock}
  }
\end{frame}

\begin{frame}{Successive Halving \citeref{Jamieson \& Talwalkar, 2016}}
  \begin{itemize}[<+->]
    \item Starts with many configurations and a small budget
    \item Iteratively trains and prunes poor-performing configurations
    \item At each stage:
    \begin{itemize}
      \item evaluate all remaining configs
      \item keep only the top fraction ($\approx \frac{1}{\eta}$)
      \item increase budget for survivors
    \end{itemize}
    \item Repeats until only a few high-quality configs remain
    \item Intuition: Spend little budget on many models at first and focus resources on the most promising ones
    \item Limitations:
    \begin{itemize}
      \item Early decisions are critical
      \item Good configurations may be discarded due to noisy low-fidelity performance
    \end{itemize}
    \item Key trade-off:
    \begin{itemize}
      \item explore many configurations with small training budget
      \item or evaluate fewer configurations more thoroughly
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Hyperband \citeref{Li et al., 2018}}
  \begin{itemize}[<+->]
    \item Addresses the need to choose a trade-off between number of configurations and training budget in Successive Halving
    \item Runs $s_{\max}+1$ Successive Halving “brackets” in parallel:
          \[
            s_{\max} = \lfloor \log_\eta (r_{\max} / r_{\min}) \rfloor
            \qquad
          \]
    \item Each bracket uses a different allocation strategy
    \begin{itemize}
    	\item From many cheap trials to few expensive ones
        \item Bracket $s{=}s_{\max}$: many configs, very cheap
        \item Bracket $s{=}0$: all configs at full budget
    \end{itemize}
    \item Covers a wide range of exploration-exploitation trade-offs simultaneously
    \item Eliminates the need to pre-select a single fidelity allocation strategy
    \item More robust: avoids failure due to a poor initial budget choice
    \item Continually restarts brackets with fresh configurations to maintain exploration
  \end{itemize}
\end{frame}

\begin{frame}{Bayesian Optimisation \citeref{Snoek et al., 2012}}
  \twocol{
    \textbf{Strategy}
    \begin{itemize}
      \item Fit a \textbf{Gaussian Process} surrogate;
            kernel: \textbf{Matérn-5/2} (handles sharp local changes
            without over-smoothing) + WhiteKernel (noise)
      \item Select next config by maximising \textbf{EI}:
    \end{itemize}
    \[
      \mathrm{EI}(\lambda) = (\mu{-}f^+{-}\xi)\,\Phi(Z) + \sigma\,\phi(Z)
    \]
    {\scriptsize $f^+$: best so far;\; $Z{=}(\mu{-}f^+{-}\xi)/\sigma$;\;
    $\xi{=}0.01$ (exploration)}
  }{
    \textbf{Implementation choices}
    \begin{itemize}
      \item Warm-start: 5 random evals before fitting GP
      \item EI maximised over pool of 1\,000 random candidates
      \item Categorical HPs: \textbf{one-hot encoded};
            log-scale HPs: \textbf{log-space normalised}
      \item Evaluates at $r_{\max}$ only (no multi-fidelity)
    \end{itemize}
  }
\end{frame}

\section{Implementation Challenges}
% ===================================================================

\begin{frame}{Implementation Challenges}
  \begin{itemize}[<+->]
    \item \textbf{Hyperband bracket restarts:}
      \begin{itemize}
        \item After restart, \texttt{tell()} routed results to wrong new bracket
        \item Stale config-to-bracket keys from the previous round were the cause
        \item Fixed by clearing the map on every restart
      \end{itemize}

    \item \textbf{BO: EI optimisation is inadequate at scale:}
      \begin{itemize}
        \item EI maximised by sampling 1\,000 random candidates
        \item In ${\sim}272$-dim one-hot space this covers negligible volume
        \item \alert{Known limitation:} a proper optimiser (e.g.\ CMA-ES)
              is needed — our BO partially degrades to random search on NB301
      \end{itemize}

    \item \textbf{BO: log-scale HP encoding bug:}
      \begin{itemize}
        \item BO was barely beating RS on rbv2 — unexpected
        \item $\mathrm{lr}{=}10^{-3}$ over $[10^{-5},1]$
              mapped to $0.001$ (linear) vs.\ ${\approx}0.4$ (correct, log)
        \item GP kernel saw most of the space as identical; fixed to log-space
      \end{itemize}

    \item \textbf{SH rung schedule — discovered post-hoc:}
      \begin{itemize}
        \item Rung progression on NB301: $[1,3,9,27,\mathbf{81}]$
        \item Never reaches $r_{\max}{=}98$ \ (98 is not a power of 3)
        \item Could not rerun; reported as known limitation in results
      \end{itemize}
  \end{itemize}
\end{frame}

\section{Conclusions}
% ===================================================================

\begin{frame}{Take-Home Messages}
  \small
  \begin{enumerate}[<+->]\setlength\itemsep{5pt}
    \item \textbf{No single best algorithm} — the winner depends on the
          benchmark landscape
    \item \textbf{Hyperband most robust:} best on NB301, matches SH on rbv2,
          lowest variance on both
    \item \textbf{Multi-fidelity wins on speed} when low-fidelity rankings
          are stable (rbv2) — early incumbent in $<$10 budget units
    \item \textbf{BO fragile at scale:} GP + random EI both degrade in
          ${\sim}272$-dim one-hot space on NB301
    \item \textbf{Implementation choices matter:} $\eta$, warm-start,
          candidate pool, and bracket-restart policy all showed up in results
  \end{enumerate}
\end{frame}

\begin{frame}{Backup: Hyperband Bracket Structure (Schematic, $\eta{=}3$)}
  Using $n_s = \lceil(B/r_{\max})\cdot\eta^s/(s+1)\rceil$,
  $\;r_s = \max(r_{\min},\, r_{\max}\cdot\eta^{-s})$.
  Example: $s_{\max}=3$, $B/r_{\max}=4$.

  \medskip
  \renewcommand{\tabcolsep}{5pt}
  \renewcommand{\arraystretch}{1.15}
  \begin{center}
  \begin{footnotesize}
  \begin{tabular}{ccccl}
    \toprule
    \textbf{Bracket} $s$ & $n_s$ & $r_s$ & Rounds & Character \\
    \midrule
    3 (most expl.) & $\lceil 4{\cdot}27/4\rceil=27$ & $r_{\min}$            & 4 & many cheap evals \\
    2              & $\lceil 4{\cdot}9/3\rceil=12$  & $r_{\min}{\cdot}\eta$ & 3 & \\
    1              & $\lceil 4{\cdot}3/2\rceil=6$   & $r_{\min}{\cdot}\eta^2$ & 2 & \\
    0 (= RS)       & $\lceil 4{\cdot}1/1\rceil=4$   & $r_{\max}$            & 1 & few full evals \\
    \bottomrule
  \end{tabular}
  \end{footnotesize}
  \end{center}
  \smallskip
  In our experiment $B/r_{\max}=50$, so $n_s$ values scale proportionally.
  Each bracket still receives $\approx B/(s_{\max}+1)$ of the total budget.
\end{frame}

\begin{frame}{Backup: EI Acquisition Function (NumPy only)}
  Given GP posterior $\mu(\lambda)$, $\sigma(\lambda)$, incumbent $f^+$,
  and exploration constant $\xi{=}0.01$:
  \[
    \mathrm{EI}(\lambda)
    = (\mu - f^+ - \xi)\,\Phi(Z)
      + \sigma\,\phi(Z),
    \qquad
    Z = \frac{\mu - f^+ - \xi}{\sigma}
  \]
  \[
    \Phi(z) = \tfrac{1}{2}\!\left[1 + \mathrm{erf}\!\left(\tfrac{z}{\sqrt{2}}\right)\right],
    \qquad
    \phi(z) = \tfrac{1}{\sqrt{2\pi}}\,e^{-z^2/2}
  \]
  \begin{itemize}
    \item Implemented with \texttt{math.erf} wrapped in \texttt{np.vectorize}
          (no scipy dependency)
    \item EI evaluated over 1\,000 random candidates per iteration;
          maximiser is selected as next query
  \end{itemize}
\end{frame}
