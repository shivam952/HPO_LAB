%% includes/macros/commenting.tex
%% Commenting and annotation macros for collaborative writing.
%% Edit the author-specific macros at the bottom to match your team.

\usepackage{xcolor}  % may already be loaded; safe to repeat

%% ── General annotations ────────────────────────────────────────────

%% \todo{text}  — marks a to-do item in red
\newcommand{\todo}[1]{%
  {\color{col-todo}\textbf{[TODO: #1]}}%
}

%% \changed{text}  — highlights changed text in orange
\newcommand{\changed}[1]{%
  {\color{col-changed}\textbf{[CHANGED: #1]}}%
}

%% \added{text}  — highlights newly added text in green
\newcommand{\added}[1]{%
  {\color{col-added}\textbf{[ADDED: #1]}}%
}
